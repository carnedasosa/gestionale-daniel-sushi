> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Creare l'app Next.js del gestionale con database, import del menù dal sito, login admin, gestione del menù da parte di Daniel (prezzi, disponibilità, **allergeni**) e menù pubblico con gli allergeni.

**Architecture:** Next.js 16 (App Router, TypeScript) con accesso al database **solo lato server** tramite il driver `postgres` sulla connessione pooler di Supabase (`DATABASE_URL`); Supabase Auth solo per il login di staff e admin. Tutta la logica (soldi, allergeni, regole d'accesso, conversione del menù, validazione) sta in moduli puri testati con Vitest; le query passano da un'interfaccia `Db` minima, così i test girano su **PGlite** (Postgres in memoria) con le stesse migrazioni SQL, senza Docker.

**Tech Stack:** Next.js 16.4 · React 19 · TypeScript · `postgres` 3.4 · `@supabase/ssr` 0.12 + `@supabase/supabase-js` 2 · Vitest 5 · `@electric-sql/pglite` 0.5 · `tsx`

**Spec:** `docs/superpowers/specs/2026-10-11-gestionale-ordini-design.md` (tappa 1 di §7)

`src/domain/soldi.ts`:
```ts
export type Centesimi = number

export function euroInCentesimi(euro: number): Centesimi {
  const cent = Math.round(euro * 100)
  if (!Number.isFinite(cent) || cent < 0) throw new RangeError(`Importo non valido: ${euro}`)
  return cent
}

// Accetta «10», «10,5», «10,50», «10.50», con o senza «€» e spazi. Niente migliaia né negativi.
const FORMATO_EURO = /^(\d{1,4})(?:[.,](\d{1,2}))?$/

export function parseEuro(testo: string): Centesimi | null {
  const pulito = testo.replace(/€/g, '').trim()
  const m = FORMATO_EURO.exec(pulito)
  if (!m) return null
  const interi = Number(m[1])
  const decimali = m[2] ? Number(m[2].padEnd(2, '0')) : 0
  return interi * 100 + decimali
}

export function formatEuro(cent: Centesimi): string {
  const euro = Math.trunc(cent / 100)
  const resto = String(Math.abs(cent % 100)).padStart(2, '0')
  return `${euro},${resto}\u00a0€`
}
```

```ts
export function eAllergene(x: string): x is CodiceAllergene {

  return ALLERGENI.find((a) => a.codice === c)!.etichetta

  for (const c of codici) if (!eAllergene(c)) throw new Error(`Allergene sconosciuto: ${c}`)
  return ALLERGENI.map((a) => a.codice).filter((c) => codici.includes(c))
}
```

```ts
import { readdirSync, readFileSync } from 'node:fs'
import path from 'node:path'

// Quello che Supabase ha già e PGlite no: schema auth, auth.uid() e i ruoli dell'API.
const STUB_SUPABASE = `
  create schema auth;
  create table auth.users (id uuid primary key, email text unique);
  create role anon nologin;
  create role authenticated nologin;
  create role service_role nologin bypassrls;
  create function auth.uid() returns uuid language sql stable as $$
    select nullif(current_setting('request.jwt.claim.sub', true), '')::uuid
  $$;
`

export async function creaDbDiTest(opzioni: { seed?: boolean } = {}): Promise<{ pg: PGlite; db: Db }> {
  const pg = new PGlite()
  await pg.exec(STUB_SUPABASE)
  const cartella = path.resolve('supabase/migrations')
  for (const file of readdirSync(cartella).filter((f) => f.endsWith('.sql')).sort()) {
```

```ts
import { euroInCentesimi } from '@/domain/soldi'

export interface MenuJson {
  categorie: Array<{
    id: string
    nome: string
    pezzi?: string
    voci: Array<{ num?: number; nome: string; prezzo: number; desc?: string; piccante?: boolean | string; veg?: boolean; casa?: boolean }>
    componi?: { titolo: string; gruppi: Array<{ nome: string; voci: string[] }> }
  }>
}

export interface RigaCategoria { id: string; nome: string; pezzi: string | null; posizione: number }

export interface RigaPiatto {
  categoria_id: string
  numero: number | null
  nome: string
  descrizione: string | null
  prezzo_cent: number
  piccante: boolean
  vegetariano: boolean
  della_casa: boolean
  posizione: number
}

export interface GruppoPoke { nome: string; voci: string[]; min: number; max: number; inclusi: number }

export interface PokeConfig {
  titolo: string
  gruppi: GruppoPoke[]
  prezzoBaseCent: number | null
  prezzoExtraCent: number | null
}

export interface MenuConvertito { categorie: RigaCategoria[]; piatti: RigaPiatto[]; poke: PokeConfig | null }

// Regole provvisorie finché Daniel non dà le sue (spec §9). Il prezzo resta null: senza prezzo la poke composta non è ordinabile.
const REGOLE_POKE_PROVVISORIE: Record<string, Pick<GruppoPoke, 'min' | 'max' | 'inclusi'>> = {
  Base: { min: 1, max: 1, inclusi: 1 },
  Proteine: { min: 1, max: 3, inclusi: 2 },
  Condimenti: { min: 0, max: 6, inclusi: 4 },
  Salse: { min: 0, max: 3, inclusi: 2 },
  Topping: { min: 0, max: 3, inclusi: 2 },
}
const REGOLA_DEFAULT = { min: 0, max: 3, inclusi: 2 }

export function convertiMenu(json: MenuJson): MenuConvertito {
  const categorie: RigaCategoria[] = []
  const piatti: RigaPiatto[] = []
  let poke: PokeConfig | null = null

  json.categorie.forEach((c, i) => {
    categorie.push({ id: c.id, nome: c.nome, pezzi: c.pezzi ?? null, posizione: i + 1 })
    c.voci.forEach((v, j) => {
      piatti.push({
        categoria_id: c.id,
        numero: v.num ?? null,
        nome: v.nome,
        descrizione: v.desc?.trim() ? v.desc.trim() : null,
        prezzo_cent: euroInCentesimi(v.prezzo),
        piccante: Boolean(v.piccante),
        vegetariano: Boolean(v.veg),
        della_casa: Boolean(v.casa),
        posizione: j + 1,
      })
    })
    if (c.componi) {
      poke = {
        titolo: c.componi.titolo,
        gruppi: c.componi.gruppi.map((g) => ({ nome: g.nome, voci: [...g.voci], ...(REGOLE_POKE_PROVVISORIE[g.nome] ?? REGOLA_DEFAULT) })),
        prezzoBaseCent: null,
        prezzoExtraCent: null,
      }
    }
  })

  return { categorie, piatti, poke }
}
```

```ts
import type { MenuConvertito } from './importa'

function lit(v: string | number | boolean | null): string {
  if (v === null) return 'null'
  if (typeof v === 'number') {
    if (!Number.isInteger(v)) throw new Error(`Numero non intero nel seed: ${v}`)
    return String(v)
  }
  if (typeof v === 'boolean') return v ? 'true' : 'false'
  return `'${v.replace(/'/g, "''")}'`
}

export function generaSeedSql(menu: MenuConvertito): string {
  const righe: string[] = ['-- Generato da scripts/genera-seed.ts: non modificare a mano.', 'begin;']

  righe.push(
    'insert into public.categorie (id, nome, pezzi, posizione) values',
    menu.categorie.map((c) => `  (${lit(c.id)}, ${lit(c.nome)}, ${lit(c.pezzi)}, ${lit(c.posizione)})`).join(',\n') + ';',
  )

  righe.push(
    'insert into public.piatti (categoria_id, numero, nome, descrizione, prezzo_cent, piccante, vegetariano, della_casa, posizione) values',
    menu.piatti
      .map((p) =>
        `  (${[p.categoria_id, p.numero, p.nome, p.descrizione, p.prezzo_cent, p.piccante, p.vegetariano, p.della_casa, p.posizione].map(lit).join(', ')})`,
      )
      .join(',\n') + ';',
  )

  if (menu.poke) {
    righe.push(
      'insert into public.poke_config (id, titolo, gruppi, prezzo_base_cent, prezzo_extra_cent) values',
      `  (1, ${lit(menu.poke.titolo)}, ${lit(JSON.stringify(menu.poke.gruppi))}::jsonb, ${lit(menu.poke.prezzoBaseCent)}, ${lit(menu.poke.prezzoExtraCent)});`,
    )
  }

  righe.push('commit;', '')
  return righe.join('\n')
}
```

```ts
  prezzo_cent: number; piccante: boolean; vegetariano: boolean; della_casa: boolean
  allergeni: CodiceAllergene[]; allergeni_confermati: boolean; disponibile: boolean; posizione: number
}

const COLONNE = `id, categoria_id, numero, nome, descrizione, prezzo_cent, piccante, vegetariano, della_casa,
  allergeni, allergeni_confermati, disponibile, posizione`

const UUID = /^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/i

function daRiga(r: RigaPiatto): Piatto {
  return {
    id: r.id, categoriaId: r.categoria_id, numero: r.numero, nome: r.nome, descrizione: r.descrizione,
    prezzoCent: r.prezzo_cent, piccante: r.piccante, vegetariano: r.vegetariano, dellaCasa: r.della_casa,
    allergeni: r.allergeni, allergeniConfermati: r.allergeni_confermati, disponibile: r.disponibile, posizione: r.posizione,
  }
}

export async function leggiMenu(db: Db): Promise<CategoriaConPiatti[]> {
  const categorie = await db.query<{ id: string; nome: string; pezzi: string | null; posizione: number }>(
    'select id, nome, pezzi, posizione from categorie order by posizione',
  )
  const piatti = await db.query<RigaPiatto>(`select ${COLONNE} from piatti order by posizione, numero nulls last`)
  return categorie.map((c) => ({ ...c, piatti: piatti.filter((p) => p.categoria_id === c.id).map(daRiga) }))
}

export async function leggiPiatto(db: Db, id: string): Promise<Piatto | null> {
  if (!UUID.test(id)) return null
  const [r] = await db.query<RigaPiatto>(`select ${COLONNE} from piatti where id = $1`, [id])
  return r ? daRiga(r) : null
}

export async function aggiornaPiatto(db: Db, id: string, d: DatiPiatto): Promise<Piatto | null> {
  if (!UUID.test(id)) return null
  const [r] = await db.query<RigaPiatto>(
    `update piatti set nome = $2, descrizione = $3, prezzo_cent = $4, piccante = $5, vegetariano = $6,
       della_casa = $7, allergeni = $8::text[], allergeni_confermati = $9, aggiornato_il = now()
     where id = $1 returning ${COLONNE}`,
    [id, d.nome, d.descrizione, d.prezzoCent, d.piccante, d.vegetariano, d.dellaCasa, d.allergeni, d.allergeniConfermati],
  )
  return r ? daRiga(r) : null
}

export async function impostaDisponibile(db: Db, id: string, disponibile: boolean): Promise<boolean> {
  if (!UUID.test(id)) return false
  const righe = await db.query('update piatti set disponibile = $2, aggiornato_il = now() where id = $1 returning id', [id, disponibile])
  return righe.length === 1
}

export async function contaAllergeniDaConfermare(db: Db): Promise<number> {
  const [{ n }] = await db.query<{ n: number }>('select count(*)::int as n from piatti where not allergeni_confermati')
  return n
}
```

```ts
export type Ruolo = 'staff' | 'admin'

function sottoSezione(pathname: string, sezione: string): boolean {
  return pathname === sezione || pathname.startsWith(`${sezione}/`)
}

export function ruoloRichiesto(pathname: string): Ruolo | null {
  if (sottoSezione(pathname, '/admin')) return 'admin'
  if (sottoSezione(pathname, '/staff')) return 'staff'
  return null
}

export function haAccesso(richiesto: Ruolo, ruolo: Ruolo | null): boolean {
  if (ruolo === null) return false
  return ruolo === 'admin' || richiesto === 'staff'
}

export function destinazioneDopoLogin(next: string | null | undefined, ruolo: Ruolo): string {
  const home = ruolo === 'admin' ? '/admin' : '/staff'
  if (!next || !next.startsWith('/') || next.startsWith('//') || next.includes('\\')) return home
  const richiesto = ruoloRichiesto(next.split(/[?#]/)[0])
  return richiesto && haAccesso(richiesto, ruolo) ? next : home
}
```

```ts
export async function entra(_stato: StatoAccesso, formData: FormData): Promise<StatoAccesso> {
  const email = String(formData.get('email') ?? '').trim()
  const password = String(formData.get('password') ?? '')
  const next = String(formData.get('next') ?? '')
  if (!email || !password) return { errore: 'Inserisci email e password.' }

  const supabase = await createClient()
  const { data, error } = await supabase.auth.signInWithPassword({ email, password })
  if (error || !data.user) return { errore: 'Email o password non corretti.' }

  const ruolo = await leggiRuolo(db, data.user.id)
  if (!ruolo) {
    await supabase.auth.signOut()
    return { errore: 'Account senza permessi: chiedi all’amministratore di abilitarlo.' }
  }
  redirect(destinazioneDopoLogin(next, ruolo))
}
```

`src/menu/valida-piatto.ts`:
```ts
import { eAllergene, normalizzaAllergeni } from '@/domain/allergeni'
import { parseEuro } from '@/domain/soldi'
import type { DatiPiatto } from './repo'

export interface InputPiatto {
  nome: string
  descrizione: string
  prezzo: string
  piccante: boolean
  vegetariano: boolean
  dellaCasa: boolean
  allergeni: string[]
  allergeniConfermati: boolean
}

type Esito = { ok: true; dati: DatiPiatto } | { ok: false; errori: Partial<Record<keyof InputPiatto, string>> }

export function validaPiatto(input: InputPiatto): Esito {
  const errori: Partial<Record<keyof InputPiatto, string>> = {}
  const nome = input.nome.trim()
  const descrizione = input.descrizione.trim()
  const prezzoCent = parseEuro(input.prezzo)

  if (!nome) errori.nome = 'Il nome è obbligatorio'
  else if (nome.length > 120) errori.nome = 'Massimo 120 caratteri'
  if (descrizione.length > 500) errori.descrizione = 'Massimo 500 caratteri'
  if (prezzoCent === null) errori.prezzo = 'Scrivi il prezzo in euro, ad esempio 10,50'
  if (!input.allergeni.every(eAllergene)) errori.allergeni = 'Allergene non valido'

  if (Object.keys(errori).length > 0) return { ok: false, errori }
  return {
    ok: true,
    dati: {
      nome,
      descrizione: descrizione || null,
      prezzoCent: prezzoCent!,
      piccante: input.piccante,
      vegetariano: input.vegetariano,
      dellaCasa: input.dellaCasa,
      allergeni: normalizzaAllergeni(input.allergeni),
      allergeniConfermati: input.allergeniConfermati,
    },
  }
}

export function inputDaForm(fd: FormData): InputPiatto {
  const testo = (k: string) => String(fd.get(k) ?? '')
  const spunta = (k: string) => fd.get(k) === 'on'
  return {
    nome: testo('nome'),
    descrizione: testo('descrizione'),
    prezzo: testo('prezzo'),
    piccante: spunta('piccante'),
    vegetariano: spunta('vegetariano'),
    dellaCasa: spunta('dellaCasa'),
    allergeni: fd.getAll('allergeni').map(String),
    allergeniConfermati: spunta('allergeniConfermati'),
  }
}
```

```ts
import { revalidatePath } from 'next/cache'
import { richiediRuolo } from '@/auth/sessione'

export type StatoModulo = { salvato: boolean; errori: Partial<Record<keyof InputPiatto, string>>; erroreGenerale: string | null }

export async function salvaPiatto(id: string, _stato: StatoModulo, fd: FormData): Promise<StatoModulo> {
  await richiediRuolo('admin')
  const esito = validaPiatto(inputDaForm(fd))
  if (!esito.ok) return { salvato: false, errori: esito.errori, erroreGenerale: null }
  const piatto = await aggiornaPiatto(db, id, esito.dati)
  if (!piatto) return { salvato: false, errori: {}, erroreGenerale: 'Piatto non trovato: forse è stato eliminato.' }
  revalidatePath('/admin/menu')
  revalidatePath('/')
  return { salvato: true, errori: {}, erroreGenerale: null }
}
```
